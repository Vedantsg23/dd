

def fibonacci_recursive(n):
    if n <= 1:
        return n
    return fibonacci_recursive(n - 1) + fibonacci_recursive(n - 2)
 
def fibonacci_iterative(term_count):
    sequence = []
    first, second = 0, 1
    for _ in range(term_count):
        sequence.append(first)
        first, second = second, first + second
    return sequence
 
term_count = int(input("How many terms? "))
if term_count <= 0:
    print("Please enter a positive integer")
else:
    iterative_sequence = fibonacci_iterative(term_count)
    recursive_sequence = [fibonacci_recursive(i) for i in range(term_count)]
    print("Iterative Fibonacci sequence:")
    for value in iterative_sequence:
        print(value)
    print("Recursive Fibonacci sequence:")
    for value in recursive_sequence:
        print(value)
