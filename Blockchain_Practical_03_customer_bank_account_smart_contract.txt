pragma solidity ^0.8.20;
contract BankAccount {
    mapping(address => uint256) private balances;
    function deposit() external payable { require(msg.value>0,"Deposit must be positive");
    balances[msg.sender]+=msg.value; }
    function withdraw(uint256 amount) external {
        require(amount>0 && balances[msg.sender]>=amount,"Insufficient balance");
        balances[msg.sender]-=amount;
        (bool ok,)=payable(msg.sender).call{value:amount}(""); require(ok,"Transfer failed");
    }
    function showBalance() external view returns(uint256) { return balances[msg.sender]; }
}
