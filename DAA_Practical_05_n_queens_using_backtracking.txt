def is_safe(board, row, column, size):
    for previous_row in range(row):
        distance = row - previous_row
        if board[previous_row][column] == 1:
            return False
        if column - distance >= 0 and board[previous_row][column - distance] == 1:
            return False
        if column + distance < size and board[previous_row][column + distance] == 1:
            return False
    return True
 
def solve_n_queens(board, row, size, backtrack_count):
    if row == size:
        return True
    for column in range(size):
        if is_safe(board, row, column, size):
            board[row][column] = 1
            if solve_n_queens(board, row + 1, size, backtrack_count):
                return True
            board[row][column] = 0
            backtrack_count[0] += 1
    return False
 
def print_board(board):
    for row in board:
        print("\t".join(str(cell) for cell in row))
 
size = int(input("Enter number of queens (N): "))
first_column = int(input(f"Enter column of first queen in row 1 (1 to {size}): "))
if size < 1 or first_column < 1 or first_column > size:
    print("Invalid input")
else:
    board = [[0 for _ in range(size)] for _ in range(size)]
    board[0][first_column - 1] = 1
    backtrack_count = [0]
    print("Matrix with first queen placed:")
    print_board(board)
    if solve_n_queens(board, 1, size, backtrack_count):
        print(f"Final {size}-Queens matrix:")
        print_board(board)
        queen_columns = tuple(row.index(1) + 1 for row in board)
        print("Column of queen in each row =", queen_columns)
        print("Number of backtracks =", backtrack_count[0])
    else:
        print(f"No solution exists with first queen at column {first_column}")
